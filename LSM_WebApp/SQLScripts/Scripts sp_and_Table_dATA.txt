CREATE PROCEDURE [dbo].[sp_ApplyLeave]
    @UserID INT, @LeaveType VARCHAR(20), @StartDate DATE, @EndDate DATE, @Reason TEXT
AS
BEGIN
    DECLARE @Days INT = DATEDIFF(DAY, @StartDate, @EndDate) + 1;
    DECLARE @Balance INT;

    -- Validation 1: Backdated Check
    IF @StartDate < CAST(GETDATE() AS DATE)
    BEGIN
        RAISERROR('Backdated leaves not allowed', 16, 1); RETURN;
    END

    -- Validation 2: Overlap Check
    IF EXISTS (SELECT 1 FROM LeaveRequests WHERE UserID = @UserID AND Status != 'Rejected' 
               AND (@StartDate <= EndDate AND @EndDate >= StartDate))
    BEGIN
        RAISERROR('Leave dates overlap with existing request', 16, 1); RETURN;
    END

    -- Validation 3: Balance Check
    SELECT @Balance = CASE 
        WHEN @LeaveType = 'Sick' THEN Sick 
        WHEN @LeaveType = 'Casual' THEN Casual 
        WHEN @LeaveType = 'Earned' THEN Earned END 
    FROM LeaveBalances WHERE UserID = @UserID;

    IF @Balance < @Days
    BEGIN
        RAISERROR('Insufficient balance', 16, 1); RETURN;
    END

    INSERT INTO LeaveRequests (UserID, LeaveType, StartDate, EndDate, Reason, Status)
    VALUES (@UserID, @LeaveType, @StartDate, @EndDate, @Reason, 'Pending');
END;
GO

CREATE PROCEDURE [dbo].[sp_ApproveRejectLeave]
    @RequestID INT, @Status VARCHAR(20), @Comment TEXT
AS
BEGIN
    UPDATE LeaveRequests SET Status = @Status, ManagerComment = @Comment WHERE RequestID = @RequestID;
    
    IF @Status = 'Approved'
    BEGIN
        DECLARE @Days INT, @LType VARCHAR(20), @UID INT;
        SELECT @Days = DATEDIFF(DAY, StartDate, EndDate)+1, @LType = LeaveType, @UID = UserID FROM LeaveRequests WHERE RequestID = @RequestID;
        
        IF @LType = 'Sick' UPDATE LeaveBalances SET Sick -= @Days WHERE UserID = @UID;
        IF @LType = 'Casual' UPDATE LeaveBalances SET Casual -= @Days WHERE UserID = @UID;
        IF @LType = 'Earned' UPDATE LeaveBalances SET Earned -= @Days WHERE UserID = @UID;
    END
END;
